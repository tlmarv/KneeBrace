Paste into Extensions → Apps Script in a blank Google Sheet, deploy as Web App (Anyone access).

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var headers = [
    "Timestamp", "Rater_ID", "Presentation_Seq", "Specimen_ID", "Filename",
    "Cat1_Axial", "Cat2_JointLine", "Cat3_AP_Plane", "Cat4_Patellar",
    "Cat5_SkinInterface", "Cat6_Straps", "Accurate_Count", "Evaluable_Count",
    "Composite_Pass", "Posture_Class", "Setting_Class", "Consensus_Flag", "Notes"
  ];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
  var data = JSON.parse(e.postData.contents);
  var row = headers.map(function(h) { return data[h] !== undefined ? data[h] : ""; });
  sheet.appendRow(row);
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
